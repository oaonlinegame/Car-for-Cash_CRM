2) แนวคิดสำคัญ (จำง่าย)

  • สร้าง Vue App แค่ครั้งเดียว ใน app.js ด้วย Vue.createApp.
  • ทุกไฟล์ JS ใช้ Vue API ได้ (เพราะโหลดผ่าน CDN) เช่น Vue.ref(), Vue.reactive().
  • สิ่งที่ Template ต้องใช้ ให้ return ออกมาจาก setup().
  • ห่อของเป็นกล่อง (namespace) เช่น const Finance = { ... } กันชื่อชนและเรียกข้ามไฟล์ง่าย.
จำสั้น ๆ: ref = .value ใน JS, reactive = ไม่ต้อง .value. ใน Template เขียนชื่อได้ตรง ๆ.

3) ref vs reactive ใช้ตอนไหน?

ref
ใช้กับค่าเดี่ยว ๆ: ตัวเลข, ข้อความ, true/false หรืออยากควบคุมการแทนที่ทั้งก้อน 
const count = Vue.ref(0)
count.value++          // ใน JS ต้อง .value
// Template: {{ count }} (ไม่ต้อง .value)
Array แบบ ref:
const list = Vue.ref([])
list.value.push('A')
list.value = []        // แทนที่ทั้งก้อน
reactive
ใช้กับ Object/Array หลายค่า (Form, Config, Data row)
const user = Vue.reactive({ name:'มาลี', age:30 })
user.age++             // ไม่ต้อง .value

const rows = Vue.reactive([])
rows.push({ id:1, name:'A' })
ข้อควรระวัง: อย่าห่อซ้อน reactive(ref(...)) จะงงง่าย เลือกแบบใดแบบหนึ่งพอ

4) แชร์ค่าข้ามไฟล์ยังไง (ไม่ต้อง import)

ทุกไฟล์ที่โหลดผ่าน <script> อยู่ global scope เหมือนกัน (หลังจากโหลด Vue แล้ว) จึงเรียกข้ามกันได้ด้วยการห่อไว้ในอ็อบเจ็กต์ชื่อไฟล์

// state.js
const Store = Vue.reactive({ user:{ name:'OAZAA', balance:100000 } })

// finance.js
const Finance = {
  loan: Vue.ref(0),
  calculate(balance){ this.loan.value = balance * 0.7 },
}

// lead.js
const Lead = {
  score(){ return Finance.loan.value > 50000 ? 'A' : 'B' },
}

// app.js
createApp({
  setup(){
    const balance = Vue.ref(0)
    return { Store, Finance, Lead, balance }
  }
}).use(createVuetify()).mount('#app')
สรุป: เรียงลำดับโหลดไฟล์ให้ถูก + ห่อเป็นอ็อบเจ็กต์ → เรียกข้ามไฟล์ได้ทันที เช่น Lead.score(), Finance.calculate(…)

5) ตัวอย่างโค้ดพร้อมใช้ (คัดลอกได้)

📄 finance.js 
// finance.js
const Finance = {
  loan: Vue.ref(0),
  calculate(balance){ this.loan.value = balance * 0.7 },
  getInstallment(months){
    const rate = 0.03
    const total = this.loan.value * (1 + rate)
    return (total / months).toFixed(2)
  }
}
📄 state.js 
// state.js
const Store = Vue.reactive({
  theme:'dark',
  user:{ name:'Guest', balance:100000 }
})
📄 app.js 
// app.js
const { createApp } = Vue
const { createVuetify } = Vuetify

createApp({
  setup(){
    const balance = Vue.ref(0)
    const compute = () => Finance.calculate(balance.value)
    return { Store, Finance, balance, compute }
  }
}).use(createVuetify()).mount('#demo')

<script src="https://cdn.jsdelivr.net/npm/vuetify@3.6.10/dist/vuetify.min.js"></script>
